from flask import Flask, render_template, request
from EmotionDetection import emotion_detector

app = Flask("Emotion Detector")

@app.route("/emotionDetector")
def send_analizer():
    text_to_analize = request.args.get('textToAnalyze')
    response = emotion_detector(text_to_analize)

    if response['dominant_emotion'] == None:
        txt = "Invalid text! Please try again!."
        return txt

    txt = f"""For the given statement, the system response is 'anger': {response['anger']}, 
    'disgust': {response['disgust']}, 'fear': {response['fear']}, 'joy': {response['joy']} and 'sadness': {response['sadness']}
    The dominant emotion is {response['dominant_emotion']}."""   
    return txt

@app.route("/")
def to_index():
    return render_template("index.html")

if __name__ == '__main__':
    app.run(host="localhost", port=5000)
